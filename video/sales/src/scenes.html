import { chidori, run, defineTool } from "chidori:agent"; const issueRefund = defineTool({ /* payments API */ }); run(async (input: { order: string; complaint: string }) => { const decision = await chidori.prompt(checkpoint `Decide a refund for ${input.order}: ${input.complaint}`); const answer = await chidori.input(`Approve? ${decision}`, {checkpoint type: "approval", choices: ["approve", "deny"] }); if (answer !== "approve") return { status: "denied" }; return await issueRefund.run(checkpoint { order: input.order, cents: 4200 }, chidori); });
